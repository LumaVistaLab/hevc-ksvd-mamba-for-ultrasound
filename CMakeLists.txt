cmake_minimum_required(VERSION 3.16)
project(usr_ultrasound LANGUAGES C)
add_executable(hevc_prior hevc_prior/encode.c hevc_prior/codec.c)
target_compile_features(hevc_prior PRIVATE c_std_99)
if(MSVC)
  target_compile_definitions(hevc_prior PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(hevc_prior PRIVATE /W3)
else()
  target_compile_options(hevc_prior PRIVATE -Wall -Wextra)
endif()
